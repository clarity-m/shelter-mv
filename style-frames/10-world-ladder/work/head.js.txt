function mulberry32(a) {
  return function () {
    a |= 0; a = a + 0x6D2B79F5 | 0;
    let t = Math.imul(a ^ a >>> 15, 1 | a);
    t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
    return ((t ^ t >>> 14) >>> 0) / 4294967296;
  };
}

  dot: (a, b) => a[0] * b[0] + a[1] * b[1] + a[2] * b[2],

// ---------------------------------------------------------------- GLSL codegen
function sdfCode(prims, name) {
  let s = `float ${name}(vec3 p, out int mat, out int part) {\n  float d = 1e9, di; mat = 0; part = 0;\n`;
  for (const q of prims) {
    let e;
    if (q.t === 'sph') e = `length(p - ${g3(q.c)}) - ${fx(q.r)}`;
    else if (q.t === 'ell') e = `sdEll(p - ${g3(q.c)}, ${g3(q.rad)})`;
    else if (q.t === 'cap') e = `sdCap(p, ${g3(q.a)}, ${g3(q.b)}, ${fx(q.r)})`;
    else if (q.t === 'rc') e = `sdRC(p, ${g3(q.a)}, ${g3(q.b)}, ${fx(q.r)}, ${fx(q.r2)})`;
    s += `  di = ${e};\n  if (di < d) { mat = ${q.mat}; part = ${q.part}; }\n  d = smin(d, di, ${fx(q.k || 0.001)});\n`;
  }
  return s + '  return d;\n}\n';
}
function sdfCodeD(prims, name) {   // distance only (shadows, AO)
  let s = `float ${name}(vec3 p) {\n  float d = 1e9;\n`;
  for (const q of prims) {
    let e;
    if (q.t === 'sph') e = `length(p - ${g3(q.c)}) - ${fx(q.r)}`;
    else if (q.t === 'ell') e = `sdEll(p - ${g3(q.c)}, ${g3(q.rad)})`;
    else if (q.t === 'cap') e = `sdCap(p, ${g3(q.a)}, ${g3(q.b)}, ${fx(q.r)})`;
    else if (q.t === 'rc') e = `sdRC(p, ${g3(q.a)}, ${g3(q.b)}, ${fx(q.r)}, ${fx(q.r2)})`;
    s += `  d = smin(d, ${e}, ${fx(q.k || 0.001)});\n`;
  }
  return s + '  return d;\n}\n';
}
function boundOf(prims) {
  let lo = [1e9, 1e9, 1e9], hi = [-1e9, -1e9, -1e9];
  const grow = (c, r) => { for (let i = 0; i < 3; i++) { lo[i] = Math.min(lo[i], c[i] - r); hi[i] = Math.max(hi[i], c[i] + r); } };
  for (const q of prims) {
    if (q.t === 'sph') grow(q.c, q.r);
    else if (q.t === 'ell') grow(q.c, Math.max(...q.rad));
    else { grow(q.a, Math.max(q.r, q.r2 || 0)); grow(q.b, Math.max(q.r, q.r2 || 0)); }
  }
  const c = v3.mul(v3.add(lo, hi), 0.5);
  return { c, r: v3.len(v3.sub(hi, lo)) * 0.5 + 0.05 };
}
