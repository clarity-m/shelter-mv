const GHOST_FS = opt => COMMON(opt) + `
uniform sampler2D uCol, uAux; uniform vec3 uSpark;
out vec4 o;
float edgeAt(ivec2 q) {
  vec4 a = texelFetch(uAux, q, 0);
  float e = 0.0;
  for (int k = 0; k < 8; k++) {
    float an = float(k) * 0.785398;
    ivec2 d = ivec2(round(vec2(cos(an), sin(an)) * 2.0));
    vec4 b = texelFetch(uAux, q + d, 0);
    if (abs(floor(b.x) - floor(a.x)) > 0.5 || abs(b.y - a.y) > 0.08 * min(a.y, b.y)) e += 0.125;
  }
  return e;
}
void main() {
  ivec2 q = ivec2(gl_FragCoord.xy);
  vec3 c = texelFetch(uCol, q, 0).rgb;
  vec4 a = texelFetch(uAux, q, 0);
  int mat = int(floor(a.x));
  vec2 px = vec2(gl_FragCoord.x, uRes.y - gl_FragCoord.y);
  if (mat == 0) { o = vec4(c, 1.0); return; }
  vec3 rd = normalize(vec3((px.x - 960.0) / F, (CY - px.y) / F, 1.0));
  vec3 p = CAM + rd * a.y;
  float e = edgeAt(q);
  float ds = length((px - uSpark.xy) / vec2(1.0, 0.8));
  float near = smoothstep(1100.0, 80.0, ds);        // how much of his light reaches here
  float ghost = mat == 11 ? 0.6 : 0.3;
  vec3 g = c * ghost;
  // silhouettes glow, warmer and brighter toward the spark
  g += (c * 0.8 + vec3(0.7, 0.45, 0.25) * near) * e * (0.8 + 1.4 * near);
  if (mat == 2 || mat == 1) {
    // the hill becomes a height map drawn in light: contour lines every 12 cm
    float hh = p.y / 0.12;
    float pw = PIXANG() * a.y / max(0.12, 0.35) * 2.2;
    float line = 1.0 - smoothstep(0.0, pw * 1.6 + 0.02, abs(fract(hh) - 0.5) * 2.0 - (1.0 - pw * 1.6));
    float d = abs(fract(hh + 0.5) - 0.5);                         // distance to the nearest contour, in steps
    float fw = PIXANG() * a.y * 9.0;                               // about a pixel, in steps
    float ln = exp(-d * d / (fw * fw + 1e-5));
    vec3 lc = mix(vec3(0.95, 0.55, 0.35), vec3(1.0, 0.82, 0.55), near);
    g += lc * ln * (0.18 + 1.1 * near * near) * smoothstep(3.2, 7.0, a.y);
  }
  if (mat >= 5 && mat <= 9) {
    // the figure: a soft inner light and a warm rim toward the spark
    vec3 n = vec3(0.0);
    g = c * 0.42 + vec3(0.9, 0.55, 0.35) * e * 1.8;
  }
  o = vec4(g, 1.0);
}`;
