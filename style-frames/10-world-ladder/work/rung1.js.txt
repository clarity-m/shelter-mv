
// ================================================================ RUNG 1: PIXEL
// The same scene ray-cast once per native pixel (480x270), read back, and drawn
// as pixel art in JS: a 26-colour palette, ramps with ordered dither only where
// bands meet, 1-px rim light marched toward the sun, and Clawd as the 36x10
// ladder sprite on the native grid (each glyph cell 1x2 native pixels, so 144x80
// on screen). Everything is then blown up 4x nearest-neighbour.
const GEO_FS = opt => COMMON(opt) + `
layout(location = 0) out vec4 G0; layout(location = 1) out vec4 G1; layout(location = 2) out vec4 G2;
void main() {
  vec2 fc = gl_FragCoord.xy;
  vec2 px = vec2(fc.x, uRes.y - fc.y) * (1920.0 / uRes.x);
  vec3 rd = normalize(vec3((px.x - 960.0) / F, (CY - px.y) / F, 1.0));
  Hit h = castScene(CAM, rd);
  float sh = 1.0, ao = 1.0;
  if (h.mat != M_SKY && h.mat != M_CLOUD) { sh = sunVis(h.p, h.n); ao = calcAO(h.p, h.n); }
  float extra = 0.0;
  if (h.mat == M_FLOOR) extra = 1.0; else if (h.mat == M_HILL) extra = floorMix(h.p.xz);
  G0 = vec4(float(h.mat) + float(h.part) / 64.0, h.t, sh, ao);
  G1 = vec4(h.n, h.fid);
  G2 = vec4(h.p, extra);
}`;
function readGBuffer(fb, w, h, n) {
  const out = [];
  gl.bindFramebuffer(gl.FRAMEBUFFER, fb);
  for (let i = 0; i < n; i++) {
    gl.readBuffer(gl.COLOR_ATTACHMENT0 + i);
    const a = new Float32Array(w * h * 4);
    gl.readPixels(0, 0, w, h, gl.RGBA, gl.FLOAT, a);
    out.push(a);
  }
  return out;
}
const PX2 = [
  '......SLLLlllllllllllllllllllS......',
  '......l###ss############ss###s......',
  '......l###oo############oo###s......',
  '......l###oo############oo###s......',
  '..Sllll#######################lllS..',
  '..ssss########################ssss..',
  '......s######################S......',
  '......SssssssssssssssssssssssS......',
  '........#s..#s........#s..#s........',
  '........sS..sS........sS..sS........',
];
const mirrorGrid = g => g.map(r => r.split('').reverse().join(''));

function rung1() {
  const NW = 480, NH = 270;
  const opt = { lowpoly: false, clawd3d: false, leafNoise: true, pal: 3 };
  const G = [tex(NW, NH, 'f32'), tex(NW, NH, 'f32'), tex(NW, NH, 'f32')];
  const fb = fbo(G);
  const prog = compile(GEO_FS(opt), 'geo1');
  timed('scene', () => run(prog, fb, {}, { uPl: plTex, uPm: pmTex }));
  const [g0, g1, g2] = timed('readback', () => readGBuffer(fb, NW, NH, 3));
  const t0 = performance.now();

  // ---------------------------------------------------------- palette (26)
  const HEX = [
    // K0..K13: dusk greys, violet in the darks, faintly warm in the lights
    '#0f0f17', '#171722', '#20202e', '#2a2a3b', '#35354a', '#43425a', '#54526b',
    '#67647d', '#7c7890', '#928da2', '#a8a2b3', '#bdb6c2', '#d2cad2', '#e6dddf',
    // S0..S4: the warm-grey glow around the sun (still unsaturated)
    '#c9bbb6', '#d9c9c0', '#e7d8cc', '#f2e6da', '#fbf5ec',
    // Clawd: # L l s S  (the only saturated colours in the world)
    '#d97757', '#f7b793', '#e8906e', '#b55b47', '#8a3f3d',
    // G0..G2: his light on the grass
    '#3b3140', '#51404a', '#6d5357',
  ];
  const RGB = HEX.map(h => [1, 3, 5].map(k => parseInt(h.slice(k, k + 2), 16)));
  const K = i => clamp(Math.round(i), 0, 13), S = i => 14 + clamp(i, 0, 4);
  const CL = { '#': 19, 'L': 20, 'l': 21, 's': 22, 'S': 23 }, GL = i => 24 + i;
  const BAY4 = [0, 8, 2, 10, 12, 4, 14, 6, 3, 11, 1, 9, 15, 7, 13, 5].map(v => (v + 0.5) / 16);
  const thr = (x, y) => BAY4[(y & 3) * 4 + (x & 3)];
  // quantise a continuous level: solid inside a band, ordered dither only near band edges
  const qd = (v, x, y, sharp) => { const b = Math.floor(v); const f = (v - b - 0.5) * (sharp || 2) + 0.5; return b + (f > thr(x, y) ? 1 : 0); };

  const IB = new Uint8Array(NW * NH);          // palette index
  const MB = new Int8Array(NW * NH);           // material
  const TB = new Float32Array(NW * NH);        // depth
  const at = (x, y) => ((NH - 1 - y) * NW + x) * 4;   // GL rows are bottom-up
  const sunPx = project(v3.add(CAM, v3.mul(SUN, 100)));
  const sN = [sunPx[0] / 4, sunPx[1] / 4];

  for (let y = 0; y < NH; y++) for (let x = 0; x < NW; x++) {
    const k = at(x, y), i = y * NW + x;
    const mat = Math.floor(g0[k]); MB[i] = mat; TB[i] = g0[k + 1];
    const d = rayDir(4 * x + 2, 4 * y + 2);
    const cs = v3.dot(d, SUN), el = Math.max(d[1], 0);
    const glow = Math.exp(-(1 - cs) * 70), wide = Math.exp(-(1 - cs) * 9);
    if (mat === MAT.SKY) {
      let L = 12.6 - 10.5 * Math.pow(el / 0.5, 0.85) + 1.1 * wide;
      if (glow > 0.3 && L > 12.3) { IB[i] = S(qd((L - 12.3) * 2.2 + glow * 1.4, x, y, 1.6)); continue; }
      IB[i] = K(qd(Math.min(L, 12.9), x, y, 1.5));
      continue;
    }
    const n = [g1[k], g1[k + 1], g1[k + 2]];
    if (mat === MAT.CLOUD) {
      const lit = v3.dot(n, SUN), under = clamp(-n[1] * 1.3, 0, 1);
      let L = 8.2 + 2.4 * smooth(-0.3, 0.6, lit) + 2.2 * under + 0.8 * wide;
      if (L > 12.2) { IB[i] = S(qd((L - 12.2) * 1.3, x, y, 2)); continue; }
      IB[i] = K(qd(L, x, y, 2.2));
      continue;
    }
    const sh = g0[k + 2], ao = g0[k + 3];
    const ndl = Math.max(v3.dot(n, SUN), 0);
    const up = clamp(n[1], 0, 1);
    const gdir = v3.norm([SUN[0], 0.12, SUN[2]]);
    const gf = Math.max(v3.dot(n, gdir), 0);
    const alb = { [MAT.HILL]: 0.3, [MAT.FLOOR]: 0.22, [MAT.TRUNK]: 0.16, [MAT.LEAF]: 0.24, [MAT.SKIN]: 0.5, [MAT.HAIR]: 0.07,
      [MAT.SWEATER]: 0.62, [MAT.PANTS]: 0.13, [MAT.SHOE]: 0.07 }[mat] || 0.3;
    const lum = alb * (2.4 * ndl * sh + 0.9 * (0.3 + 0.7 * up) * ao + 0.9 * gf * gf * ao);
    // map luminance to the ramp: a gentle log so darks keep separation
    let L = 1.2 + 9.5 * Math.pow(clamp(lum, 0, 1.4), 0.62);
    if (mat === MAT.HILL || mat === MAT.FLOOR) {
      const fm = g2[k + 3];
      L = lerp(L, 3.2, fm * 0.6);
      L += (hash2(Math.floor(g2[k] * 2.5), Math.floor(g2[k + 2] * 2.5)) - 0.5) * 0.7;   // grass patches
    }
    IB[i] = K(qd(L, x, y, 2.2));
  }
  function hash2(a, b) { let h = (a * 374761393 + b * 668265263) | 0; h = (h ^ (h >>> 13)) * 1274126177; return ((h ^ (h >>> 16)) >>> 0) / 4294967296; }

  // ---------------------------------------------------------- dome grid: 1-px lines, one step lighter
  const domeIdx = (x, y) => {
    const d = rayDir(4 * x + 2, 4 * y + 2);
    const oc = v3.sub(CAM, DOME.c), b = v3.dot(oc, d), c = v3.dot(oc, oc) - DOME.R * DOME.R;
    const t = -b + Math.sqrt(b * b - c);
    const q = v3.mul(v3.sub(v3.add(CAM, v3.mul(d, t)), DOME.c), 1 / DOME.R);
    const st = 10 * Math.PI / 180;
    return [Math.floor(Math.atan2(q[0], q[2]) / st), Math.floor(Math.asin(q[1]) / st), Math.asin(q[1])];
  };
  const DI = [];
  for (let y = 0; y < NH; y++) for (let x = 0; x < NW; x++) DI.push(domeIdx(x, y));
  for (let y = 0; y < NH - 1; y++) for (let x = 0; x < NW - 1; x++) {
    const i = y * NW + x;
    if (MB[i] !== MAT.SKY || DI[i][2] < 0.02) continue;
    const a = DI[i], r = DI[i + 1], b = DI[i + NW];
    const line = a[0] !== r[0] || a[1] !== b[1];
    if (!line) continue;
    if ((x + y) & 1) continue;                                // dotted: faint
    const v = IB[i];
    if (v <= 12) IB[i] = Math.min(v + 1, 13);
  }

  // ---------------------------------------------------------- sun disc and stepped halo
  for (let y = 0; y < NH; y++) for (let x = 0; x < NW; x++) {
    const i = y * NW + x;
    if (MB[i] !== MAT.SKY) continue;
    const r = Math.hypot(x + 0.5 - sN[0], y + 0.5 - sN[1]);
    if (r < 6.6) IB[i] = S(4);
    else if (r < 9.5 + thr(x, y) * 1.2) IB[i] = Math.max(IB[i] >= 14 ? IB[i] : 0, S(3));
  }

  // ---------------------------------------------------------- rim light: 1 px on edges facing the sun
  const sx = sN[0], sy = sN[1];
  const RIMC = { [MAT.HILL]: 9, [MAT.LEAF]: 8, [MAT.TRUNK]: 6, [MAT.SWEATER]: 12, [MAT.SKIN]: 11, [MAT.HAIR]: 7, [MAT.PANTS]: 6, [MAT.SHOE]: 5 };
  const IB2 = IB.slice();
  for (let y = 1; y < NH - 1; y++) for (let x = 1; x < NW - 1; x++) {
    const i = y * NW + x, m = MB[i];
    if (m === MAT.SKY || m === MAT.CLOUD || m === MAT.FLOOR || !(m in RIMC)) continue;
    // step one pixel toward the sun (rounded to the dominant axis)
    const dx = sx - x, dy = sy - y, l = Math.hypot(dx, dy);
    const ux = Math.abs(dx / l) > 0.38 ? Math.sign(dx) : 0, uy = Math.abs(dy / l) > 0.38 ? Math.sign(dy) : 0;
    const j = (y + uy) * NW + (x + ux), ju = (y - 1) * NW + x;
    const open = (jj) => MB[jj] === MAT.SKY || MB[jj] === MAT.CLOUD || (TB[jj] > TB[i] * 1.25 && MB[jj] !== m);
    if (open(j) || (m !== MAT.HILL && open(ju) && uy <= 0)) {
      const near = Math.exp(-Math.hypot(x - sx, y - sy) / 140);
      IB2[i] = K(Math.max(IB[i], RIMC[m] + (near > 0.35 ? 1 : 0)));
    }
  }
  IB.set(IB2);

  // ---------------------------------------------------------- grass tufts on the skyline
  const rnd = mulberry32(4242);
  for (let x = 1; x < NW - 1; x++) {
    let top = -1;
    for (let y = 0; y < NH; y++) { const m = MB[y * NW + x]; if (m === MAT.HILL) { top = y; break; } if (m !== MAT.SKY && m !== MAT.CLOUD) break; }
    if (top < 2) continue;
    const r = rnd();
    const hgt = r < 0.2 ? 2 : r < 0.55 ? 1 : 0;
    for (let k = 1; k <= hgt; k++) IB[(top - k) * NW + x] = IB[top * NW + x];
  }

  // ---------------------------------------------------------- Clawd: 36x10 sprite, light from the right
  const cp = project([CLAWD.x, CLAWD.y, CLAWD.z]);
  const cx = Math.round(cp[0] / 4), cyN = Math.round(cp[1] / 4);
  const spr = mirrorGrid(PX2);
  const x0 = cx - 18, y0 = cyN - 20;
  // his light: warm rings in the air, a warm pool on the grass below him
  const LX = cx, LY = cyN - 10;
  for (let y = Math.max(0, LY - 40); y < Math.min(NH, LY + 40); y++) for (let x = Math.max(0, LX - 60); x < Math.min(NW, LX + 60); x++) {
    const i = y * NW + x, m = MB[i];
    const r = Math.hypot((x + 0.5 - LX) / 1.35, y + 0.5 - LY);
    const lv = 2.6 * Math.pow(clamp(1 - r / 34, 0, 1), 1.25);
    const q = qd(lv, x, y, 2);
    if (q < 1) continue;
    if (m === MAT.SKY || m === MAT.CLOUD) { if (IB[i] < 14) IB[i] = q >= 2 ? S(1) : S(0); else IB[i] = Math.min(IB[i] + 1, S(4)); }
    else if (m === MAT.HILL) IB[i] = GL(Math.min(q, 2));
  }
  for (let r = 0; r < 10; r++) for (let c = 0; c < 36; c++) {
    const ch = spr[r][c];
    if (ch === '.' || ch === 'o') continue;
    for (let k = 0; k < 2; k++) { const x = x0 + c, y = y0 + r * 2 + k; if (x >= 0 && y >= 0 && x < NW && y < NH) IB[y * NW + x] = CL[ch]; }
  }
  TIMES.pixelJS = performance.now() - t0;

  // ---------------------------------------------------------- 4x nearest
  const img = octx.createImageData(W, H), D = img.data;
  for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
    const c = RGB[IB[(y >> 2) * NW + (x >> 2)]], k = (y * W + x) * 4;
    D[k] = c[0]; D[k + 1] = c[1]; D[k + 2] = c[2]; D[k + 3] = 255;
  }
  octx.putImageData(img, 0, 0);
  log('rung1 colours used: ' + new Set(IB).size);
}
